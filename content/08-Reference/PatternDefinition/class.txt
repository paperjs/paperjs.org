<div class="reference-class">


<p>A pattern definition repeats an item as the tile of pattern colors, which fill and stroke items with it, see <a href="/reference/color#pattern"><tt>color.pattern</tt></a>: a <a href="/reference/raster"><tt>Raster</tt></a> for image patterns, or any other item for vector patterns. As with symbol definitions, the item is removed from the scene graph, and all pattern colors that use the pattern share it, so they change along with it.</p>
<h4>Example:<span class="description">A dotted pattern, filling and stroking a circle:</span></h4>

<div class="paperscript split">

<div class="buttons">
<div class="button run">Run</div>
</div>

<script type="text/paperscript" canvas="canvas-0">
var dot = new Path.Circle({
    center: [5, 5],
    radius: 3,
    fillColor: 'red'
});

// Repeat the dot every 10 points, with a gap of 2 points between the
// tiles of the stroke's pattern:
var pattern = new PatternDefinition(dot, {
    tile: new Rectangle(0, 0, 10, 10)
});
var spaced = new PatternDefinition(dot.clone(), {
    tile: new Rectangle(0, 0, 10, 10),
    spacing: 2
});

var circle = new Path.Circle({
    center: view.center,
    radius: 70,
    fillColor: { pattern: pattern },
    strokeColor: {
        pattern: spaced,
        matrix: new Matrix().rotate(45)
    },
    strokeWidth: 20
});
</script>
